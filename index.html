   ========================================= */

/* Famílias Base */
h1, h2, h3, h4, h5, h6, p, li, span, .rd-button { font-family: "Open Sans", sans-serif; color: #14142C; }

/* Headers com Atitude (Anton) */
h1, h1 *, h2, h2 * { 
    font-family: "Anton", sans-serif !important; 
    text-transform: uppercase !important; 
    letter-spacing: 0.02em !important;
    line-height: 1.1 !important;
}
h1, h1 * { font-size: 64px !important; color: #0A00BD !important; }
h2, h2 * { font-size: 48px !important; color: #14142C !important; }

/* Subtítulos (Kanit) */
h3, h3 *, h4, h4 * { font-family: "Kanit", sans-serif !important; font-weight: 700 !important; }
h3, h3 * { font-size: 36px !important; color: #FF3399 !important; text-transform: uppercase !important;}
h4, h4 * { font-size: 28px !important; }

/* Botões Nativos RD (Padrão Primário Responsivo) */
.rd-button {
    font-family: "Kanit", sans-serif !important; font-weight: 700 !important; text-transform: uppercase !important;
    padding: 16px 32px !important; background-color: #FF3399 !important; color: #14142C !important;
    border: 2px solid transparent !important; border-radius: 0px !important; box-shadow: none !important;
    transition: all 0.2s cubic-bezier(0.25, 0.8, 0.25, 1) !important;
    width: 100% !important; /* Full width em mobile por default */
}
@media(min-width: 768px) { .rd-button { width: auto !important; } }
.rd-button:hover { background-color: #FFB001 !important; transform: translate(-4px, -4px) !important; box-shadow: 6px 6px 0px #14142C !important; }

/* Correções Mobile Nativas */
@media(max-width: 767px) {
    h1, h1 * { font-size: 40px !important; }
    h2, h2 * { font-size: 32px !important; }
    h3, h3 * { font-size: 28px !important; }
}
